```
type Color = 'Black' | 'White';
type X = 'A' | 'B' | 'C'; // литералы типов
type Y = 1 | 2 | 3 | 4;

class Position {
  constructor(
    private xPos: X, // модификатор доступа private автоматически присваивает параметр к this
    // только экземпляр Position может считывать из приватного параметра и производить в него запись
    private yPos: Y
  ) {}

  distanceFrom(position: Position) {
    return {
      xPos: Math.abs(position.xPos.charCodeAt(0) - this.xPos.charCodeAt(0)),
      yPos: Math.abs(position.yPos - this.yPos)
    }
  }
}

abstract class Piece {
  protected position: Position // protected привязывает свойство к this 
  // protected видим для экземпляра Piece И для экземпляров любых ПОДКЛАССОВ Piece

  constructor(
    private readonly color: Color,
    // private: присвоение this + доступен только из класса Piece, 
    // readonly: после изначального присваивания доступность только для чтения
    xPos: X,
    yPos: Y
  ) {
    this.position = new Position(xPos, yPos)
  }

  moveTo(position: Position){ // по умолчанию public, предустановленная реализация
    this.position = position
  }
  abstract canMoveTo(position: Position): boolean // подкласс обязан создать метод совместимый с этой сигнатурой
}

// abstract: нельзя сделать непосредственный инстанс, только расширять

class King extends Piece {
  canMoveTo(position: Position) {
     let distance = this.position.distanceFrom(position);
     return distance.xPos < 2 && distance.yPos < 2
  }
}

class Game {
  private pieces = Game.makePieces()

  private static makePieces() {
    return [
      new King('White', 'A', 1),
      new King('Black', 'A', 1)
    ]
  }
}
```
