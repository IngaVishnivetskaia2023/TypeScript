#### Nominal vs Structural Typing
```
THE CORE DIFFERENCE:
════════════════════

NOMINAL TYPING (Java, C#):
──────────────────────────
  "Types are identified by their NAME"

  class Dog { name: string }
  class Cat { name: string }
  
  Dog ≠ Cat   (different names = different types)
  Even though they have IDENTICAL structure


STRUCTURAL TYPING (TypeScript ⭐):
──────────────────────────────────
  "Types are identified by their SHAPE"

  interface Dog { name: string }
  interface Cat { name: string }
  
  Dog === Cat  ✅ (same shape = compatible)
  TypeScript doesn't care about the NAME
```

#### The never Type

```
MENTAL MODEL:
═════════════

  Think of types as SETS of possible values:

  string  = { "hello", "world", "", "abc", ... }  infinite set
  number  = { 0, 1, -3, 3.14, ... }               infinite set
  boolean = { true, false }                        2 values
  null    = { null }                               1 value
  undefined = { undefined }                        1 value
  
  unknown = EVERYTHING (universal set)  ← top type
  never   = NOTHING (empty set)         ← bottom type

  ┌─────────────── unknown ──────────────────┐
  │  ┌─────────┐  ┌────────┐  ┌──────────┐  │
  │  │ string  │  │ number │  │ boolean  │  │
  │  └─────────┘  └────────┘  └──────────┘  │
  │  ┌──────┐  ┌───────────┐                │
  │  │ null │  │ undefined │                │
  │  └──────┘  └───────────┘                │
  │                                          │
  │            never = ∅ (empty)             │
  │            (exists nowhere, fits nowhere)│
  └──────────────────────────────────────────┘
```


```
// WHERE never APPEARS:

// 1. Functions that NEVER return
function throwError(msg: string): never {
  throw new Error(msg);    // never reaches end
}

function infiniteLoop(): never {
  while (true) { }         // never reaches end
}


// 2. Exhaustive switch checks
type Shape = "circle" | "square" | "triangle";

function area(shape: Shape) {
  switch (shape) {
    case "circle":    return /* ... */;
    case "square":    return /* ... */;
    case "triangle":  return /* ... */;
    default:
      // shape is `never` here — all cases handled
      const _exhaustive: never = shape;
      // If someone adds "pentagon" to Shape,
      // this line ERRORS at compile time ⭐
      // "Type 'pentagon' is not assignable to type 'never'"
      return _exhaustive;
  }
}


// 3. Impossible intersections
type Impossible = string & number; // never
// Nothing can be BOTH string AND number


// 4. Filtered conditional types
type NonString<T> = T extends string ? never : T;
type Result = NonString<string | number | boolean>;
// = never | number | boolean
// = number | boolean  (never disappears from unions)
```

#### Arrays, Tuples, Objects, Records
```
// ═══ ARRAYS — variable length, single type ═══

const names: string[] = ["Alice", "Bob", "Charlie"];
const names2: Array<string> = ["Alice", "Bob"]; // same thing
names.push("Dave"); // ✅ can grow


// ═══ TUPLES — fixed length, specific type per position ═══

const point: [number, number] = [10, 20];
//            pos 0   pos 1

const response: [number, string, boolean] = [200, "OK", true];
//               status  message  success

// NAMED TUPLES (documentation only — no runtime effect)
type APIResponse = [status: number, body: string, ok: boolean];
const res: APIResponse = [200, '{"data":1}', true];
// Helps with readability in editor tooltips ⭐


// ═══ READONLY variants ═══

const fixed: readonly number[] = [1, 2, 3];
fixed.push(4);  // ❌ Property 'push' does not exist
fixed[0] = 99;  // ❌ Index signature only permits reading

const fixedTuple: readonly [string, number] = ["age", 25];
fixedTuple[0] = "name"; // ❌ Cannot assign to '0'
```

#### `as const` — The Deep Freeze

```

// ═══ as const = make EVERYTHING readonly + literal ═══

// WITHOUT as const:
const config = {
  api: "https://api.com",
  retries: 3,
  methods: ["GET", "POST"]
};
// Type: { api: string, retries: number, methods: string[] }
//       Widened types — "string" not "https://api.com"


// WITH as const:
const config = {
  api: "https://api.com",
  retries: 3,
  methods: ["GET", "POST"]
} as const;
// Type: {
//   readonly api: "https://api.com",      ← literal
//   readonly retries: 3,                  ← literal
//   readonly methods: readonly ["GET", "POST"]  ← readonly tuple
// }

config.api = "x";           // ❌ readonly
config.methods.push("PUT"); // ❌ readonly
config.retries = 5;         // ❌ readonly


// COMMON PATTERN: extract union from as const

const ROLES = ["admin", "user", "guest"] as const;
type Role = (typeof ROLES)[number];
// = "admin" | "user" | "guest"

// Instead of manually keeping union + array in sync!


// ═══ as const vs Readonly vs ReadonlyArray ═══

// ReadonlyArray<T> — shallow, just array methods
const a: ReadonlyArray<number> = [1, 2, 3];
a.push(4);  // ❌
// But if items are objects, their properties are mutable

// Readonly<T> — shallow, just top-level properties
type Config = Readonly<{ api: string; nested: { port: number } }>;
// { readonly api: string; readonly nested: { port: number } }
// nested.port is still MUTABLE ⚠️

// as const — DEEP readonly + literal types
// Everything frozen, all the way down ⭐
```

#### Call Signatures & Construct Signatures

```
// ═══ CALL SIGNATURE — how to type a callable ═══

// Shorthand:
type Greet = (name: string) => string;

// Full (allows adding properties):
type Greet = {
  (name: string): string;        // call signature
  defaultLang: string;           // property on function
};

const greet: Greet = (name) => `Hello ${name}`;
greet.defaultLang = "en";
greet("Alice");         // ✅ callable
greet.defaultLang;      // ✅ has property


// ═══ CONSTRUCT SIGNATURE — typing `new` ═══

interface UserConstructor {
  new (name: string, age: number): User;  // construct signature
}

// "This can be called with `new` and returns a User"

class User {
  constructor(public name: string, public age: number) {}
}

function createUser(Ctor: UserConstructor): User {
  return new Ctor("Alice", 30);  // new is allowed because
}                                // of construct signature

createUser(User); // ✅


// ═══ BOTH call + construct signature ═══

interface DateConstructor {
  new (): Date;           // new Date()
  (): string;             // Date() — without new
  parse(s: string): number;
}
```


#### bind vs Inline Arrow, strictBindCallApply

```
// ═══ THE this PROBLEM ═══

class Button {
  label = "Click me";

  handleClick() {
    console.log(this.label); // `this` depends on HOW it's called
  }
}

const btn = new Button();

// PROBLEM:
document.addEventListener("click", btn.handleClick);
// `this` = document, not btn! 💥 → undefined


// ═══ THREE FIXES ═══

// FIX 1: bind — creates new function with fixed `this`
document.addEventListener("click", btn.handleClick.bind(btn));
// Pros: explicit, reusable
// Cons: creates new function reference each time

// FIX 2: Arrow function in class field
class Button2 {
  label = "Click me";
  handleClick = () => {        // arrow captures `this` at creation
    console.log(this.label);   // `this` is ALWAYS Button2 instance
  };
}
// Pros: automatic, clean
// Cons: new function per instance (memory), not on prototype

// FIX 3: Inline arrow wrapper
document.addEventListener("click", () => btn.handleClick());
// Pros: simple
// Cons: new function each render (React re-render issue)


// ═══ strictBindCallApply ═══

// tsconfig: "strictBindCallApply": true

function greet(name: string, age: number): string {
  return `${name} is ${age}`;
}

// WITH strictBindCallApply: true
greet.call(null, "Alice", 30);      // ✅ correct args
greet.call(null, "Alice", "thirty"); // ❌ arg 2 must be number
greet.bind(null, "Alice")(30);       // ✅ partial application typed
greet.apply(null, ["Alice", 30]);    // ✅ array form

// WITHOUT (default): all args typed as `any` — no safety ⚠️
```

#### Static block
```
// ═══ STATIC BLOCK — run code during class initialization ═══

class Database {
  static connection: string;
  static isReady: boolean;

  // Static block — runs ONCE when class is loaded
  static {
    // Complex initialization logic
    try {
      Database.connection = loadConnectionString();
      Database.isReady = true;
      console.log("DB initialized");
    } catch {
      Database.connection = "fallback://localhost";
      Database.isReady = false;
    }
  }
}

// WHY use static block?
// ✅ Access to private static fields
// ✅ Try/catch during initialization
// ✅ Complex multi-step setup
// ✅ Runs exactly ONCE at class evaluation time
// Before this, you'd need IIFE or separate init function


// MULTIPLE static blocks run in order:
class Config {
  static value: number;
  
  static { Config.value = 1; }
  
  someProperty = "hello";  // instance property (between blocks)
  
  static { Config.value += 10; }
  
  // Config.value = 11 after class loads
}
```

#### Auto Accessor
```
// ═══ AUTO ACCESSOR — shorthand for get/set with private storage ═══

class User {
  accessor name: string;  // ← auto accessor
  
  constructor(name: string) {
    this.name = name;
  }
}

// WHAT IT COMPILES TO (conceptually):

class User {
  #__name: string;  // hidden private storage

  get name(): string {
    return this.#__name;
  }
  
  set name(value: string) {
    this.#__name = value;
  }

  constructor(name: string) {
    this.name = name;
  }
}

// WHY use auto accessor?
// ✅ Works with decorators (decorators need get/set, not fields)
// ✅ Can later add validation without changing public API
// ✅ Less boilerplate than manual get/set

// ═══ WITH DECORATORS ═══
function validate(target: any, context: ClassAccessorDecoratorContext) {
  return {
    set(value: string) {
      if (!value) throw new Error("Name required");
      target.set.call(this, value);
    },
    get() {
      return target.get.call(this);
    }
  };
}

class User {
  @validate
  accessor name: string = "";  // decorator can intercept get/set ✅
}

// Without `accessor`, decorators can't intercept plain fields
// because fields don't have get/set at the language level
```

#### Conditional Types

```
// ═══ SYNTAX: T extends U ? TrueType : FalseType ═══

// Like a ternary, but for TYPES

type IsString<T> = T extends string ? "yes" : "no";

type A = IsString<string>;   // "yes"
type B = IsString<number>;   // "no"
type C = IsString<"hello">;  // "yes" (literal extends string)


// ═══ DISTRIBUTIVE behavior with unions ═══

type IsString<T> = T extends string ? "yes" : "no";

type D = IsString<string | number>;
// Distributes! Becomes:
// IsString<string> | IsString<number>
// = "yes" | "no"

// PREVENT distribution with [T]:
type IsStringNonDist<T> = [T] extends [string] ? "yes" : "no";
type E = IsStringNonDist<string | number>; // "no" (union as whole)


// ═══ PRACTICAL EXAMPLES ═══

// Extract only string keys
type StringKeys<T> = {
  [K in keyof T]: T[K] extends string ? K : never;
}[keyof T];

interface User { name: string; age: number; email: string; }
type UserStringKeys = StringKeys<User>; // "name" | "email"


// Flatten promise
type Unwrap<T> = T extends Promise<infer U> ? U : T;
type A = Unwrap<Promise<string>>; // string
type B = Unwrap<number>;          // number
```


#### The infer Keyword
```
// ═══ infer = "capture this type for me" ═══

// Mental model: infer is like a VARIABLE in a conditional type
// "If T matches this pattern, EXTRACT the part I marked with infer"

// Extract return type of a function:
type ReturnOf<T> = T extends (...args: any[]) => infer R
  ? R      // R is whatever the return type was
  : never;

type A = ReturnOf<() => string>;           // string
type B = ReturnOf<(x: number) => boolean>; // boolean


// Extract element type from array:
type ElementOf<T> = T extends (infer E)[] ? E : never;

type C = ElementOf<string[]>;        // string
type D = ElementOf<(number | boolean)[]>; // number | boolean


// Extract first argument:
type FirstArg<T> = T extends (first: infer F, ...rest: any[]) => any
  ? F
  : never;

type E = FirstArg<(name: string, age: number) => void>; // string


// Multiple infer:
type ParseRoute<T> = T extends `${infer Start}/:${infer Param}/${infer Rest}`
  ? { start: Start; param: Param; rest: Rest }
  : never;

type Route = ParseRoute<"api/:userId/posts">;
// { start: "api"; param: "userId"; rest: "posts" }


// ═══ infer with constraints (TS 4.7+) ═══

type FirstString<T> = T extends [infer F extends string, ...any[]]
  ? F       // F is guaranteed to be string
  : never;
```

### Advanced Utility Types
#### ThisType, ThisParameterType, OmitThisParameter, InstanceType

```
// ═══ ThisType<T> — define what `this` means inside an object ═══

// PROBLEM: You have a config object where methods
// should access other properties via `this`

interface AppState {
  count: number;
  name: string;
}

// ThisType tells TS: "inside these methods, `this` = AppState"
const store: ThisType<AppState> & {
  increment(): void;
  reset(): void;
} = {
  increment() {
    this.count++;    // ✅ TS knows this.count exists
    this.name;       // ✅ TS knows this.name exists
  },
  reset() {
    this.count = 0;  // ✅ fully typed
  }
};

// REAL WORLD: Vue 2 Options API uses this pattern!
// methods, computed, data all share `this` context


// ═══ ThisParameterType<T> — extract the `this` type ═══

function greet(this: { name: string }, greeting: string) {
  return `${greeting}, ${this.name}`;
}

type GreetThis = ThisParameterType<typeof greet>;
// = { name: string }

type NoThis = ThisParameterType<(x: number) => void>;
// = unknown (no explicit this parameter)


// ═══ OmitThisParameter<T> — remove `this` from function type ═══

type GreetFn = typeof greet;
// = (this: { name: string }, greeting: string) => string

type CleanGreetFn = OmitThisParameter<typeof greet>;
// = (greeting: string) => string
// `this` parameter removed!

// USE CASE: When passing method as callback
// where `this` will be bound separately
const boundGreet: CleanGreetFn = greet.bind({ name: "Alice" });
boundGreet("Hello"); // ✅


// ═══ InstanceType<T> — get instance type from class/constructor ═══

class UserService {
  name = "service";
  getUser(id: number) { return { id, name: "Alice" }; }
}

type UserServiceInstance = InstanceType<typeof UserService>;
// = UserService (the instance type)

// WHY useful? When you receive a CLASS (constructor), not instance:
function createService<T extends new (...args: any[]) => any>(
  ServiceClass: T
): InstanceType<T> {
  return new ServiceClass();
}

const service = createService(UserService);
// TypeScript knows: service is UserService instance
service.getUser(1); // ✅ fully typed
```
